Command-Line Interface

# Browse model catalog
open-higgsfield-ai models --category t2v

# CinemaStudio camera compilation (Dry Run)
open-higgsfield-ai cinema \
  --prompt "Neon rain in Neo Tokyo" \
  --camera modular8k \
  --lens anamorphic \
  --focal-length 35 \
  --aperture "f/1.4" \
  --lighting noir \
  --pan 15 \
  --dry-run

# Run Embedded Web Studio & REST Server
open-higgsfield-ai serve --host 127.0.0.1 --port 3000

Rust Crate Dependency

[dependencies]
open-higgsfield-ai = "0.0.1"
tokio = { version = "1.39", features = ["full"] }

// Native Rust Client Usage:
use open_higgsfield_ai::{HiggsfieldClient, CinemaStudio, CinemaStudioRequest};

let client = HiggsfieldClient::from_env()?;
let studio = CinemaStudio::new(&client);
